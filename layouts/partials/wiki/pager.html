{{- $currentPage := . -}} {{- $prev := false -}} {{- $next := false -}} {{- $wikiPages := slice -}} {{- range (where .Site.Pages "Section" "wiki") -}} {{- if and .Params.icon (ne .Params.icon "") -}} {{- $wikiPages = $wikiPages | append . -}} {{- end -}} {{- end -}} {{- $sorted := sort $wikiPages "Weight" "asc" -}} {{- range $index, $p := $sorted -}} {{- if eq $p.RelPermalink $currentPage.RelPermalink -}} {{- if gt $index 0 -}} {{- $prev = index $sorted (sub $index 1) -}} {{- end -}} {{- if lt (add $index 1) (len $sorted) -}} {{- $next = index $sorted (add $index 1) -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{- with $prev }} Previous {{ .Title }} {{- else }}
{{- end }} {{- with $next }} Next {{ .Title }} {{- else }}
{{- end }}
{{- end -}}
